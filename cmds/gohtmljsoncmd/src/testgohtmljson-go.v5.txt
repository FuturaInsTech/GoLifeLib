// Used in testgohtmljson processing
func formatInteger(famt string, ctype byte) string {
	// Handle the Indian and Western numbering formats
	n := len(famt)
	if n <= 3 {
		// No formatting needed for numbers with 3 or fewer digits
		return famt
	}

	// Format the first three digits (thousands)
	result := famt[n-3:]

	if ctype == 'a' || ctype == 'A' {
		// Format the remaining digits in groups of two (lakhs, crores, etc.)
		for i := n - 3; i > 0; i -= 2 {
			start := i - 2
			if start < 0 {
				start = 0
			}
			result = famt[start:i] + "," + result
		}
	}
	if ctype == 'c' || ctype == 'C' {
		// Format the remaining digits in groups of three (millions, billions, etc.)
		for i := n - 3; i > 0; i -= 3 {
			start := i - 3
			if start < 0 {
				start = 0
			}
			result = famt[start:i] + "," + result
		}
	}

	// Return the formatted integer
	return result
}

func formatDecimal(famt string, ctype byte) string {
	decimalpart := ""
	decimallen, _ := strconv.Atoi(string(ctype))
	//damt, _ := strconv.Atoi(string(famt))
	var decimalFloat float64 = 0
	if decimallen == 0 {
		decimalFloat, _ = strconv.ParseFloat("0."+"0", 64)
	} else {
		decimalFloat, _ = strconv.ParseFloat("0."+string(famt), 64)
	}
	if decimallen == 0 {
		return decimalpart
	} else {
		formatString := fmt.Sprintf("%%.%df", decimallen) // Generate format string
		decimalpart = fmt.Sprintf(formatString, decimalFloat)
		decimalpart = decimalpart[1:] // Remove the leading "0"
		return decimalpart
	}
}

func formatNumber(value interface{}, fds string) (interface{}, error) {
	integralpart := ""
	decimalpart := ""
	famt := ""
	famount := ""
	decimallen, _ := strconv.Atoi(string(fds[1]))
	if _, ok := value.(float64); ok {
		famt = strconv.FormatFloat(value.(float64), 'f', decimallen, 64) // Format float64 with d decimal places
	}
	if _, ok := value.(int); ok {
		famt = strconv.FormatFloat(float64(value.(int)), 'f', decimallen, 64) // Format float64 with d decimal places
	}
	if _, ok := value.(string); ok {
		famt = value.(string)
	}
	parts := strings.Split(famt, ".")

	switch fds[0] {
	case 'a', 'A':
		integralpart = formatInteger(parts[0], fds[0])
	case 'c', 'C':
		integralpart = formatInteger(parts[0], fds[0])
	case 'd', 'D':
		integralpart = parts[0]
	default:
		integralpart = parts[0]
	}

	if len(parts) > 1 {
		decimalpart = formatDecimal(parts[1], fds[1])
		famount = integralpart + decimalpart
	} else {
		famount = integralpart
	}
	return fmt.Sprintf("%s", famount), nil // Format float64 with 2 decimal places and comma separators
}

// formatValue is to determine which formatting function to call
// func formatValue(value interface{}, fds string) error {
func formatValue(value interface{}, fds string) (interface{}, error) {

	fds = strings.TrimSpace(fds)

	if fds == "" {
		return "", fmt.Errorf("format string cannot be empty")
	}

	switch fds[0] {

	case 'a', 'A',
		'c', 'C',
		'd', 'D':

		result, _ := formatNumber(value, fds)

		return result, nil

	case 't', 'T':

		// Table processing is handled separately
		// after the field name is identified.
		return value, nil

	default:
		return "", fmt.Errorf("Incorrect Format Type - %q", fds)
	}
}

// Custom template functions
func CreateFuncMap() template.FuncMap {
	return template.FuncMap{
		// NOTE:
		// The existing gohtml files generated used formatNumber as key.
		// As the impact is HIGH retain the key to be the same though it calls formatValue.
		// After gohtml generator is changed from formatNumber to formatValue,
		// and existing gohtml files are also corrected,
		// this key shall also be changed as formatValue.
		"formatNumber": formatValue,
		"contains":     strings.Contains, // <C>.<Field> STR.text.</C> Tag
		"eq": func(a, b interface{}) bool { // <E>.<Field> STR.text..</E> Tag
			return fmt.Sprintf("%v", a) == fmt.Sprintf("%v", b)
		},
		"ne": func(a, b interface{}) bool { // <N>.<Field> STR.text..</N> Tag
			return fmt.Sprintf("%v", a) != fmt.Sprintf("%v", b)
		},
		"in": func(val interface{}, options ...interface{}) bool { // <I>.<Field> [S T R].text..</I> Tag
			valStr := fmt.Sprintf("%v", val)
			for _, opt := range options {
				if valStr == fmt.Sprintf("%v", opt) {
					return true
				}
			}
			return false
		},
		"out": func(val string, options ...interface{}) bool { // <O>.<Field> [S T R].text...</O> Tag
			valStr := fmt.Sprintf("%v", val)
			for _, opt := range options {
				if valStr == fmt.Sprintf("%v", opt) {
					return false
				}
			}
			return true
		},
	}
}

func testgohtmljson(gohtmlFile string, jsonFile string, outFile string, iFile string) {
	// Read the HTML file
	jsonData, err := os.ReadFile(jsonFile)
	if err != nil {
		fmt.Printf("Error reading JSON file: %v\n", err)
		return
	}

	// Parse JSON into a map
	var data map[string]interface{}
	err = json.Unmarshal(jsonData, &data)
	if err != nil {
		log.Fatalf("Error parsing JSON: %v", err)
	}

	// Get the current working directory
	cwdPath, err := os.Getwd()
	if err != nil {
		fmt.Println("Error:", err)
		return
	}

	iPath := filepath.Join(cwdPath, "static")
	imagePath := filepath.Join(iPath, iFile)

	// Read the gohtml file
	fileContent, err := os.ReadFile(gohtmlFile)
	if err != nil {
		fmt.Println("Error reading file:", err)
		return
	}

	// Convert content to string and replace {{.Img}} before parsing
	modContent := strings.ReplaceAll(string(fileContent), "{{.Img}}", imagePath)

	// Process All Table expressions to inject CSS and format HTML content
	if err = processTableExpressions(&modContent, data); err != nil {
		log.Fatalf("process table HTML update and/or table CSS injection error: %w", err)
	}

	modifiedContent := modContent

	// Parse and execute template
	funcMap := CreateFuncMap()
	tmpl, err := template.New(gohtmlFile).Funcs(funcMap).Parse(modifiedContent)
	if err != nil {
		log.Fatalf("Error loading template: %v", err)
	}

	// Render template to standard output or file
	outputFile, err := os.Create(outFile)
	if err != nil {
		log.Fatalf("Error creating output file: %v", err)
	}
	defer outputFile.Close()

	err = tmpl.Execute(outputFile, data)
	if err != nil {
		log.Fatalf("Error executing template: %v", err)
	}

	fmt.Printf("Test outfile created successfully at: %s\n", outFile)
}

/*
=====================================================================
PROCESS TABLE DATA FORMATTING OF GOHTML TEMPLATE BEGINS HERE
=====================================================================

ProcessTable performs all table-related processing.
Parameters:
	htmlContent
	    Pointer to the existing HTML document.
	fieldName
	    Name of the field used in the GoHTML.
	    Example:
	    "otable_data"
	tableValue
	    Actual value of the table field.
	    Supported values:
	    - map[string]interface{}
	    - JSON string
	    - []byte containing JSON
	    - other JSON-compatible values
	format
	    Table formatting instruction.
	    Examples:
	    "t4"
	    "T4"
	    "t10"
	    t4 means:
	        t = table
	        4 = maximum 4 data columns
The function:
	1. Validates the table format.
	2. Generates the exact GoHTML placeholder.
	3. Generates the table HTML.
	4. Replaces the placeholder.
	5. Generates table CSS.
	6. Injects CSS into the existing <style> tag.
The supplied htmlContent is modified in-place.
===========================================================
*/

func ProcessTable(htmlContent *string, fieldName string, tableValue interface{}, format string, orientation string, originalExpression string) error {

	if htmlContent == nil {
		return fmt.Errorf("htmlContent cannot be nil")
	}

	/*
		=======================================================
		1. DETERMINE TABLE SIZE
		=======================================================
	*/

	columnCount, isTable := parseTableFormat(format)

	if !isTable {
		return fmt.Errorf("invalid table format %q: expected tN or TN", format)
	}

	/*
		=======================================================
		2. BUILD EXACT GOHTML PLACEHOLDER
		=======================================================

		For:

		    fieldName = otable_data
		    format    = t4

		we generate:

		    {{ formatNumber $.otable_data "t4" }}
	*/

	// placeholder := buildPlaceholder(fieldName, format, orientation)
	placeholder := originalExpression

	/*
		=======================================================
		3. CONVERT TABLE DATA
		=======================================================
	*/

	data, err := toJSONValue(tableValue) // for all type of fields

	if err != nil {
		return fmt.Errorf("unable to convert table data: %w", err)
	}

	/*
		=======================================================
		4. GENERATE TABLE HTML (all field types supported)
		=======================================================
	*/
	tableHTML := buildTableHTMLForValue(data, fieldName, columnCount, orientation)

	/*
		=======================================================
		5. REPLACE PLACEHOLDER
		=======================================================
	*/

	if !strings.Contains(
		*htmlContent,
		placeholder,
	) {

		return fmt.Errorf("table placeholder not found: %s", placeholder)
	}

	*htmlContent = strings.Replace(*htmlContent, placeholder, tableHTML, 1)

	/*
		=======================================================
		6. GENERATE TABLE CSS
		=======================================================
	*/

	tableCSS := buildTableCSS()

	/*
		=======================================================
		7. INJECT CSS INTO <style>
		=======================================================

		Because ProcessTable can be called multiple times,
		injectCSS() checks whether the table CSS is already
		present.

		Therefore:

		    First table -> CSS inserted

		    Second table -> CSS not duplicated

		    Third table -> CSS not duplicated
	*/

	updatedHTML, err := injectCSS(*htmlContent, tableCSS)

	if err != nil {
		return err
	}

	*htmlContent = updatedHTML

	return nil
}

/*
===========================================================
PARSE TABLE FORMAT
===========================================================

Examples:

	t4   -> table, 4 columns
	T4   -> table, 4 columns
	t10  -> table, 10 columns
	t15  -> table, maximum 10 columns

Invalid:

	x4
	x10
	t
	t0
	abc

Maximum allowed = 10.
===========================================================
*/

func parseTableFormat(format string) (int, bool) {

	format =
		strings.TrimSpace(format)

	if len(format) < 2 {
		return 0, false
	}

	/*
		-------------------------------------------------------
		First character must be t or T.
		-------------------------------------------------------
	*/

	if format[0] != 't' &&
		format[0] != 'T' {

		return 0, false
	}

	/*
		-------------------------------------------------------
		Remaining characters must be numeric.
		-------------------------------------------------------
	*/

	count, err :=
		strconv.Atoi(format[1:])

	if err != nil {
		return 0, false
	}

	if count <= 0 {
		return 0, false
	}

	/*
		-------------------------------------------------------
		Maximum table size = 10.
		-------------------------------------------------------
	*/

	// if count > 10 {
	// 	count = 10
	// }

	return count, true
}

/*
===========================================================
GET TABLE TITLE
===========================================================

Reads:

	"TableGridTitle": "Insurers/Features"

The title is used as the first table header.

If the field is absent, "" is used.
===========================================================
*/

func getTableTitle(data map[string]interface{}) string {
	// TFH4 is the preferred inner grid title. Keep the older
	// TableGridTitle key working for backward compatibility.
	const defaultGridTitle = ""
	if gridtitle := getMetadataString(data, "TFH4"); gridtitle != "" {
		return gridtitle
	}
	if gridtitle := getMetadataString(data, "TableGridTitle"); gridtitle != "" {
		return gridtitle
	}
	return defaultGridTitle
}

func getOuterHeading(data map[string]interface{}) string {
	return getMetadataString(data, "TFH1")
}

func getOuterSubHeading(data map[string]interface{}) string {
	return getMetadataString(data, "TFH2")
}

func getInnerTableHeading(data map[string]interface{}) string {
	// Support the previous key too, so older JSON still renders.
	if heading := getMetadataString(data, "TFH3"); heading != "" {
		return heading
	}
	return getMetadataString(data, "TableHeading")
}

func getInnerFooterHeading(data map[string]interface{}) string {
	if heading := getMetadataString(data, "TFH5"); heading != "" {
		return heading
	}
	return getMetadataString(data, "FooterHeading")
}

func getOuterFooter(data map[string]interface{}) string {
	return getMetadataString(data, "TFH6")
}

func getMetadataString(data map[string]interface{}, key string) string {
	value, exists := data[key]
	if !exists || value == nil {
		return ""
	}
	text, ok := value.(string)
	if !ok {
		return ""
	}
	return strings.TrimSpace(text)
}

func isTableMetadata(key string) bool {
	switch key {
	case "TFH1", "TFH2", "TFH3", "TFH4", "TFH5", "TFH6",
		"Heading", "SubHeading", "TableHeading", "TableGridTitle", "FooterHeading", "FootNotes":
		return true
	default:
		return false
	}
}

/*
===========================================================
GENERIC JSON TABLE SUPPORT
===========================================================

Supported values:

	scalar       -> label/value table
	[]any        -> vertical or horizontal 1-D table
	[][]any      -> matrix; *_hor transposes the matrix
	map[string]any -> key/value table, vertical or horizontal
	map-of-maps  -> existing comparison-table format

The orientation is inferred from the field name suffix _ver or _hor.
If neither suffix is present, vertical is used.
*/
func toJSONValue(value interface{}) (interface{}, error) {
	switch v := value.(type) {
	case string:
		var decoded interface{}
		if err := json.Unmarshal([]byte(v), &decoded); err == nil {
			return decoded, nil
		}
		return v, nil
	case []byte:
		var decoded interface{}
		if err := json.Unmarshal(v, &decoded); err != nil {
			return nil, err
		}
		return decoded, nil
	default:
		raw, err := json.Marshal(value)
		if err != nil {
			return nil, err
		}
		var decoded interface{}
		if err := json.Unmarshal(raw, &decoded); err != nil {
			return nil, err
		}
		return decoded, nil
	}
}

func buildTableHTMLForValue(value interface{}, fieldName string, maxColumns int, orientation string) string {
	// Keep the established comparison-table renderer for map-of-map data.
	if obj, ok := value.(map[string]interface{}); ok {
		// Optional metadata wrapper: {"TFH1": "...", "data": ...}.
		if payload, exists := obj["data"]; exists {
			meta := map[string]interface{}{}
			for k, v := range obj {
				if k != "data" {
					meta[k] = v
				}
			}
			return buildSimpleTableHTML(payload, maxColumns, orientation, meta)
		}
		for key, item := range obj {
			if isTableMetadata(key) {
				continue
			}
			if _, isObject := item.(map[string]interface{}); isObject {
				return buildTableHTML(obj, maxColumns)
			}
		}
	}
	return buildSimpleTableHTML(value, maxColumns, orientation, nil)
}

func buildSimpleTableHTML(value interface{}, maxColumns int, orientation string, metadata map[string]interface{}) string {
	if orientation == "" {
		orientation = "v" // backward-compatible default when third argument is omitted
	}

	var b strings.Builder
	if metadata != nil {
		if h := getOuterHeading(metadata); h != "" {
			b.WriteString(`<div class="tabledata-outer-heading">` + escapeHTML(h) + `</div>` + "\n")
		}
		if h := getOuterSubHeading(metadata); h != "" {
			b.WriteString(`<div class="tabledata-outer-subheading">` + escapeHTML(h) + `</div>` + "\n")
		}
	}
	b.WriteString(`<table class="tabledata-table">` + "\n")

	switch v := value.(type) {
	case []interface{}:
		if isMapArray(v) {
			// 2D map array: map keys are column titles; each map is a data row.
			columnSet := make(map[string]bool)
			for _, item := range v {
				rowMap, _ := item.(map[string]interface{})
				for key := range rowMap {
					columnSet[key] = true
				}
			}
			columns := make([]string, 0, len(columnSet))
			for key := range columnSet {
				columns = append(columns, key)
			}
			sort.Strings(columns)
			if len(columns) > maxColumns {
				columns = columns[:maxColumns]
			}
			b.WriteString("<thead><tr>")
			for _, key := range columns {
				b.WriteString(`<th class="tabledata-column">` + escapeHTML(key) + `</th>`)
			}
			b.WriteString("</tr></thead><tbody>\n")
			for _, item := range v {
				rowMap, _ := item.(map[string]interface{})
				b.WriteString("<tr>")
				for _, key := range columns {
					b.WriteString(`<td class="tabledata-value">` + escapeHTML(valueToString(rowMap[key])) + `</td>`)
				}
				b.WriteString("</tr>\n")
			}
			b.WriteString("</tbody>\n")
		} else if is2DArray(v) {
			rows := make([][]interface{}, 0, len(v))
			for _, item := range v {
				row, _ := item.([]interface{})
				rows = append(rows, row)
			}
			if orientation == "h" {
				rows = transposeRows(rows)
			}
			b.WriteString("<tbody>\n")
			for _, row := range rows {
				b.WriteString("<tr>")
				limit := len(row)
				if limit > maxColumns {
					limit = maxColumns
				}
				for i := 0; i < limit; i++ {
					b.WriteString(`<td class="tabledata-value">` + escapeHTML(valueToString(row[i])) + `</td>`)
				}
				b.WriteString("</tr>\n")
			}
			b.WriteString("</tbody>\n")
		} else if orientation == "h" {
			// 1D array: values only, no synthetic column headings.
			b.WriteString("<tbody><tr>")
			limit := len(v)
			if limit > maxColumns {
				limit = maxColumns
			}
			for i := 0; i < limit; i++ {
				b.WriteString(`<td class="tabledata-value">` + escapeHTML(valueToString(v[i])) + `</td>`)
			}
			b.WriteString("</tr></tbody>\n")
		} else {
			// 1D array: values only, one value per row.
			limit := len(v)
			if limit > 100 {
				limit = 100
			}
			b.WriteString("<tbody>\n")
			for _, item := range v[:limit] {
				b.WriteString(`<tr><td class="tabledata-value">` + escapeHTML(valueToString(item)) + `</td></tr>` + "\n")
			}
			b.WriteString("</tbody>\n")
		}
	case map[string]interface{}:
		keys := make([]string, 0, len(v))
		for k := range v {
			keys = append(keys, k)
		}
		sort.Strings(keys)
		if orientation == "h" {
			if len(keys) > maxColumns {
				keys = keys[:maxColumns]
			}
			// Map keys are meaningful titles, so keep them as headers.
			b.WriteString("<thead><tr>")
			for _, k := range keys {
				b.WriteString(`<th class="tabledata-column">` + escapeHTML(k) + `</th>`)
			}
			b.WriteString("</tr></thead><tbody><tr>")
			for _, k := range keys {
				b.WriteString(`<td class="tabledata-value">` + escapeHTML(valueToString(v[k])) + `</td>`)
			}
			b.WriteString("</tr></tbody>\n")
		} else {
			// A 1D map is a key/value table;
			limit := len(keys)
			if limit > 100 {
				limit = 100
			}
			for _, k := range keys[:limit] {
				b.WriteString(`<tr><td class="tabledata-row-heading">` + escapeHTML(k) + `</td><td class="tabledata-value">` + escapeHTML(valueToString(v[k])) + `</td></tr>` + "\n")
			}
			b.WriteString("</tbody>\n")
		}
	default:
		// Scalar: exactly one cell containing the value, with no label/header.
		b.WriteString(`<tbody><tr><td class="tabledata-value">` + escapeHTML(valueToString(v)) + `</td></tr></tbody>` + "\n")
	}
	b.WriteString("</table>\n")
	if metadata != nil {
		if h := getInnerFooterHeading(metadata); h != "" {
			b.WriteString(`<div class="tabledata-outer-subheading">` + escapeHTML(h) + `</div>` + "\n")
		}
		if h := getOuterFooter(metadata); h != "" {
			b.WriteString(`<div class="tabledata-outer-footer">` + escapeHTML(h) + `</div>` + "\n")
		}
	}
	return b.String()
}

func isMapArray(v []interface{}) bool {
	if len(v) == 0 {
		return false
	}
	_, ok := v[0].(map[string]interface{})
	return ok
}

func is2DArray(v []interface{}) bool {
	if len(v) == 0 {
		return false
	}
	_, ok := v[0].([]interface{})
	return ok
}

func transposeRows(rows [][]interface{}) [][]interface{} {
	max := 0
	for _, row := range rows {
		if len(row) > max {
			max = len(row)
		}
	}
	out := make([][]interface{}, max)
	for c := 0; c < max; c++ {
		out[c] = make([]interface{}, 0, len(rows))
		for _, row := range rows {
			if c < len(row) {
				out[c] = append(out[c], row[c])
			} else {
				out[c] = append(out[c], nil)
			}
		}
	}
	return out
}

/*
===========================================================
GET TABLE COLUMNS
===========================================================

TableGridTitle is metadata and is NOT considered a row.

Column names are collected from the inner objects.
===========================================================
*/

func getColumns(data map[string]interface{}, maxColumns int) []string {

	columnSet :=
		make(map[string]bool)

	for rowName, rowValue := range data {
		/*
			---------------------------------------------------
			TableGridTitle is not a table row.
			---------------------------------------------------
		*/

		if isTableMetadata(rowName) {
			continue
		}

		row, ok :=
			rowValue.(map[string]interface{})

		if !ok {
			continue
		}

		for columnName := range row {
			columnSet[columnName] = true
		}
	}

	columns := make([]string, 0, len(columnSet))

	for columnName := range columnSet {
		columns = append(columns, columnName)
	}

	/*
		-------------------------------------------------------
		Sort for deterministic output.
		-------------------------------------------------------
	*/

	sort.Strings(columns)

	/*
		-------------------------------------------------------
		Apply tN maximum.
		-------------------------------------------------------
	*/

	if len(columns) > maxColumns {
		columns = columns[:maxColumns]
	}

	return columns
}

/*
===========================================================
BUILD TABLE HTML
===========================================================
*/

func buildTableHTML(data map[string]interface{}, maxColumns int) string {
	outerHeading := getOuterHeading(data)             // TFH1: centered above table
	outerSubHeading := getOuterSubHeading(data)       // TFH2: left aligned above table
	innerHeading := getInnerTableHeading(data)        // TFH3: centered row inside table
	tableTitle := getTableTitle(data)                 // TFH4: normal grid header cell
	innerFooterHeading := getInnerFooterHeading(data) // TFH5: final row inside table
	outerFooter := getOuterFooter(data)               // TFH6: left aligned below table

	columns := getColumns(data, maxColumns)
	colspan := strconv.Itoa(len(columns) + 1) // row-label column plus product columns

	var b strings.Builder

	// Outer headings are deliberately outside the <table> element.
	if outerHeading != "" {
		b.WriteString(`<div class="tabledata-outer-heading">`)
		b.WriteString(escapeHTML(outerHeading))
		b.WriteString("</div>\n")
	}
	if outerSubHeading != "" {
		b.WriteString(`<div class="tabledata-outer-subheading">`)
		b.WriteString(escapeHTML(outerSubHeading))
		b.WriteString("</div>\n")
	}

	b.WriteString(`<table class="tabledata-table">` + "\n<thead>\n")

	// TFH3: optional inner heading spanning every table column.
	if innerHeading != "" {
		b.WriteString(`<tr class="tabledata-table-heading"><th colspan="` + colspan + `">`)
		b.WriteString(escapeHTML(innerHeading))
		b.WriteString("</th></tr>\n")
	}

	// TFH4 (or legacy TableGridTitle): first cell in the regular grid header.
	b.WriteString(`<tr><th class="tabledata-heading">`)
	b.WriteString(escapeHTML(tableTitle))
	b.WriteString("</th>\n")
	for _, column := range columns {
		b.WriteString(`<th class="tabledata-column">`)
		b.WriteString(escapeHTML(column))
		b.WriteString("</th>\n")
	}
	b.WriteString("</tr>\n</thead>\n<tbody>\n")

	rowNames := make([]string, 0, len(data))
	for rowName := range data {
		if isTableMetadata(rowName) {
			continue
		}
		if _, ok := data[rowName].(map[string]interface{}); ok {
			rowNames = append(rowNames, rowName)
		}
	}
	sort.Strings(rowNames)

	for _, rowName := range rowNames {
		row := data[rowName].(map[string]interface{})
		b.WriteString(`<tr><td class="tabledata-row-heading">`)
		b.WriteString(escapeHTML(rowName))
		b.WriteString("</td>\n")
		for _, column := range columns {
			value := interface{}(nil)
			if cellValue, exists := row[column]; exists {
				value = cellValue
			}
			b.WriteString(`<td class="tabledata-value">`)
			b.WriteString(escapeHTML(valueToString(value)))
			b.WriteString("</td>\n")
		}
		b.WriteString("</tr>\n")
	}
	b.WriteString("</tbody>\n")

	// TFH5: optional footer row inside the table, left aligned.
	if innerFooterHeading != "" {
		b.WriteString(`<tfoot><tr class="tabledata-footer-heading"><td colspan="` + colspan + `">`)
		b.WriteString(escapeHTML(innerFooterHeading))
		b.WriteString("</td></tr></tfoot>\n")
	}
	b.WriteString("</table>\n")

	// TFH6: optional reference/note outside and below the table.
	if outerFooter != "" {
		b.WriteString(`<div class="tabledata-outer-footer">`)
		b.WriteString(escapeHTML(outerFooter))
		b.WriteString("</div>\n")
	}
	return b.String()
}

/*
===========================================================
CONVERT CELL VALUE TO STRING
===========================================================
*/

func valueToString(value interface{}) string {

	if value == nil {
		return "-"
	}

	switch v := value.(type) {
	case string:
		if strings.TrimSpace(v) == "" {
			return "-"
		}
		return v

	case float64:
		if v == float64(int64(v)) {
			return strconv.FormatInt(int64(v), 10)
		}

		return strconv.FormatFloat(v, 'f', -1, 64)

	case float32:
		return strconv.FormatFloat(float64(v), 'f', -1, 32)

	case bool:
		return strconv.FormatBool(v)

	default:
		return fmt.Sprintf("%v", value)
	}
}

/*
===========================================================
INJECT CSS INTO EXISTING <style> TAG
===========================================================

Important:

ProcessTable() can be called multiple times.

Therefore this function first checks whether the
.tabledata-table CSS already exists.

First call:
    CSS is inserted.

Second call:
    CSS is not inserted again.

Third call:
    CSS is not inserted again.
===========================================================
*/

func injectCSS(htmlContent string, css string) (string, error) {
	/*
		-------------------------------------------------------
		Do not inject the same CSS more than once.
		-------------------------------------------------------
	*/

	if strings.Contains(htmlContent, ".tabledata-table {") {

		return htmlContent, nil
	}

	/*
		-------------------------------------------------------
		Find </style>.
		-------------------------------------------------------
	*/

	lowerHTML :=
		strings.ToLower(htmlContent)

	styleEnd :=
		strings.Index(lowerHTML, "</style>")

	if styleEnd == -1 {

		return "",
			fmt.Errorf("existing </style> tag not found")
	}

	/*
		-------------------------------------------------------
		Insert generated CSS immediately before </style>.
		-------------------------------------------------------
	*/

	return htmlContent[:styleEnd] +
			"\n" +
			css +
			"\n" +
			htmlContent[styleEnd:],
		nil
}

// escapeHTML
func escapeHTML(s string) string {
	replacer := strings.NewReplacer(
		"&", "&amp;",
		"<", "&lt;",
		">", "&gt;",
		`"`, "&#34;",
		"'", "&#39;",
	)
	return replacer.Replace(s)
}

// process table formatting fields to insert the style and replace table fields with html codes
func processTableExpressions(htmlContent *string, data map[string]interface{}) error {

	if htmlContent == nil {
		return fmt.Errorf("html content cannot be nil")
	}

	/*
		-------------------------------------------------------
		Find expressions such as:

		{{ formatNumber $.otable_data "t4" }}

		{{ formatNumber $.another_table "t3" "h" }}

		{{ formatNumber $.another_table "t3" "v" }}
		-------------------------------------------------------
	*/

	re := regexp.MustCompile(
		//`\{\{\s*formatNumber\s+\$\.(\w+)\s+"([tT]\d+)"\s*\}\}`,
		//`\{\{\s*formatNumber\s+\$\.(\w+)\s+"([tT]\d+)"(?:\s+"([hvHV])")?\s*\}\}`,
		`\{\{\s*formatNumber\s+\$\.(\w+)\s+"([tT]\d+)"(?:\s+"([hvHV])")?\s*\}\}`,
	)

	matches := re.FindAllStringSubmatch(*htmlContent, -1)

	for _, match := range matches {

		if len(match) != 4 {
			continue
		}

		/*
			---------------------------------------------------
			match[0] = original formatting expresssion
			match[1] = actual field name
			match[2] = formatting instruction
			match[3] = orientation (h or v)
			---------------------------------------------------
		*/
		originalExpression := match[0]
		fieldName := match[1]
		format := match[2]
		orientation := strings.ToLower(match[3])

		// Default orientation for existing 2D Map formatting.
		if orientation == "" {
			orientation = "h"
		}

		value, exists := data[fieldName]

		if !exists {
			return fmt.Errorf("table field %q not found in template data", fieldName)
		}

		err := ProcessTable(htmlContent, fieldName, value, format, orientation, originalExpression)

		if err != nil {
			return fmt.Errorf(
				"processing table %q: %w",
				fieldName,
				err,
			)
		}
	}

	return nil
}
